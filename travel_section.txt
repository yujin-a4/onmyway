    const ROUTE_POINTS = [
      { x: 690, y: 150 }, { x: 650, y: 200 }, { x: 610, y: 235 }, { x: 575, y: 275 },
      { x: 535, y: 330 }, { x: 490, y: 380 }, { x: 455, y: 410 }, { x: 440, y: 450 }, { x: 430, y: 505 }
    ];
    function distToRoute(px, py) {
      let minD = Infinity;
      for (const p of ROUTE_POINTS) {
        const d = Math.hypot(px - p.x, py - p.y);
        if (d < minD) minD = d;
      }
      return minD;
    }
    function pinColorByDist(dist) {
      if (dist < 80) return { border: '#059669', bg: '#ecfdf5' }; // 초록
      if (dist < 180) return { border: '#f59e0b', bg: '#fffbeb' }; // 노랑
      return { border: '#ef4444', bg: '#fff1f2' };    // 빨강
    }

    let _wpPopup = null;
    function editWaypoint(pinEl, wpId) {
      // 기존 팝업 닫기
      if (_wpPopup) { _wpPopup.remove(); _wpPopup = null; }

      const labelEl = document.getElementById(wpId);
      const currentText = labelEl.textContent.replace(/^[🏠🏢]\s*/, '');

      const popup = document.createElement('div');
      popup.className = 'waypoint-edit-popup';

      // 핀 기준 위치 (map-vp 안에서)
      const pinRect = pinEl.getBoundingClientRect();
      const vpRect = document.getElementById('map-vp').getBoundingClientRect();
      const popLeft = Math.min(pinRect.left - vpRect.left - 10, vpRect.width - 230);
      const popTop = pinRect.top - vpRect.top + 10;

      popup.style.left = Math.max(8, popLeft) + 'px';
      popup.style.top = popTop + 'px';

      const icon = wpId === 'wp-0' ? '🏠' : '🏢';
      popup.innerHTML = `
    <h4>거점 이름 변경</h4>
    <input id="wp-input" type="text" value="${currentText}" placeholder="예: 집, 회사, 학교…" maxlength="12"/>
    <div class="waypoint-edit-actions">
      <button class="wp-btn-cancel" onclick="closeWpPopup()">취소</button>
      <button class="wp-btn-save"   onclick="saveWaypoint('${wpId}','${icon}')">저장</button>
    </div>`;

      document.getElementById('map-vp').appendChild(popup);
      _wpPopup = popup;

      const inp = document.getElementById('wp-input');
      inp.focus(); inp.select();
      inp.addEventListener('keydown', e => {
        if (e.key === 'Enter') saveWaypoint(wpId, icon);
        if (e.key === 'Escape') closeWpPopup();
      });
    }

    }

    function initNamingMap(step) {
      const type = selectedUserType || '직장인';
      const config = ROUTE_CONFIG[type] || ROUTE_CONFIG['직장인'];
      const route = config[step];
      const container = document.getElementById('kakao-map-naming');
      if (!container) return;
      if (typeof kakao === 'undefined' || !kakao.maps) {
        container.style.background = 'linear-gradient(160deg,#1a1a2e 0%,#16213e 60%,#0f3460 100%)';
        return;
      }
      try {
        const midPt = route.path[Math.floor(route.path.length / 2)];
        if (!namingMap) {
          namingMap = new kakao.maps.Map(container, {
            center: new kakao.maps.LatLng(midPt[0], midPt[1]), level: 7
          });
        } else {
          namingMap.setCenter(new kakao.maps.LatLng(midPt[0], midPt[1]));
          namingMap.setLevel(7);
        }
        namingPolys.forEach(p => p.setMap(null));
        namingPolys = [];
        const poly = new kakao.maps.Polyline({
          path: route.path.map(([lat, lng]) => new kakao.maps.LatLng(lat, lng)),
          strokeWeight: 8, strokeColor: route.color, strokeOpacity: 0.95, strokeStyle: 'solid'
        });
        poly.setMap(namingMap);
        namingPolys.push(poly);
        if (route.busPath) {
          const busPoly = new kakao.maps.Polyline({
            path: route.busPath.map(([lat, lng]) => new kakao.maps.LatLng(lat, lng)),
            strokeWeight: 6, strokeColor: '#f97316', strokeOpacity: 0.9, strokeStyle: 'shortdot'
          });
          busPoly.setMap(namingMap);
          namingPolys.push(busPoly);
        }
      } catch(e) {
        container.style.background = 'linear-gradient(160deg,#1a1a2e 0%,#16213e 60%,#0f3460 100%)';
        console.warn('namingMap init failed:', e);
      }
    }

    function rnNext() {
      const val = document.getElementById('rn-input').value.trim();
      if (!val) return;
      const type = selectedUserType || '직장인';
      const config = ROUTE_CONFIG[type] || ROUTE_CONFIG['직장인'];
      const total = config.length;
      routeNames[rnStep] = val;
      if (rnStep < total - 1) {
        rnStep++;
        const dots = document.querySelectorAll('.rn-step-dot');
        dots.forEach((d, i) => d.classList.toggle('active', i === rnStep));
        const curEl = document.getElementById('rn-cur');
        const badgeEl = document.getElementById('rn-badge');
        const inputEl = document.getElementById('rn-input');
        const btnEl = document.getElementById('rn-btn');
        if (curEl) curEl.textContent = String(rnStep + 1);
        if (badgeEl) badgeEl.textContent = config[rnStep].from + ' → ' + config[rnStep].to;
        if (inputEl) inputEl.value = '';
        if (btnEl) {
          btnEl.classList.remove('active');
          btnEl.textContent = rnStep === total - 1 ? '내 지도 보기' : '다음 노선 보기 →';
        }
        initNamingMap(rnStep);
      } else {
        goTo('s-map');
        mapInit();
      }
    }

    function toggleRoute(i) {
      const onCount = routeOnStates.filter(v => v).length;
      if (routeOnStates[i] && onCount <= 1) return;
      routeOnStates[i] = !routeOnStates[i];
      const type = selectedUserType || '직장인';
      const config = ROUTE_CONFIG[type] || ROUTE_CONFIG['직장인'];
      if (routePolylines[i]) {
        routePolylines[i].forEach(p => p.setMap(routeOnStates[i] ? kakaoMap : null));
      }
      const btn = document.getElementById('rb-' + i);
      if (btn) {
        if (routeOnStates[i]) {
          btn.style.background = config[i].color;
          btn.style.color = '#fff';
        } else {
          btn.style.background = 'rgba(255,255,255,0.92)';
          btn.style.color = '#9ca3af';
        }
      }
    }

    // ── 노선 거리 기반 색상 ──
    function haverDist(lat1, lng1, lat2, lng2) {
      const R = 6371000;
      const dLat = (lat2 - lat1) * Math.PI / 180;
      const dLng = (lng2 - lng1) * Math.PI / 180;
      const a = Math.sin(dLat/2)**2 + Math.cos(lat1*Math.PI/180) * Math.cos(lat2*Math.PI/180) * Math.sin(dLng/2)**2;
      return R * 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a));
    }
    function distToRoutePath(lat, lng, path) {
      let minD = Infinity;
      for (const [plat, plng] of path) {
        const d = haverDist(lat, lng, plat, plng);
        if (d < minD) minD = d;
      }
      return minD;
    }
    function pinColorByDist(dist) {
      if (dist < 500)  return { border: '#059669', bg: '#ecfdf5' };  // 초록
      if (dist < 1500) return { border: '#f59e0b', bg: '#fffbeb' };  // 노랑
      return               { border: '#ef4444', bg: '#fff1f2' };     // 빨강
    }

    // ── mapInit (사용자 유형별 동적 노선) ──
    function mapInit() {
      const container = document.getElementById('kakao-map');
      if (!container) return;
      if (typeof kakao === 'undefined' || !kakao.maps) {
        container.style.background = '#1e1e2e';
        container.style.display = 'flex';
        container.style.alignItems = 'center';
        container.style.justifyContent = 'center';
        container.innerHTML = '<div style="color:#a78bfa;text-align:center;padding:24px;font-size:14px;font-weight:700">' +
          '📍 카카오 지도 초기화 실패<br><span style="font-size:12px;color:#9ca3af;font-weight:400;margin-top:8px;display:block">' +
          '카카오 콘솔 → 앱 설정 → 플랫폼 → Web에<br><b style="color:#a78bfa">http://localhost:52861</b> 도메인을 등록해주세요</span></div>';
        return;
      }

      const type = selectedUserType || '직장인';
      const config = ROUTE_CONFIG[type] || ROUTE_CONFIG['직장인'];

      // 지도 중심 계산
      const allPts = config.flatMap(r => [...r.path, ...(r.busPath || [])]);
      const avgLat = allPts.reduce((s, p) => s + p[0], 0) / allPts.length;
      const avgLng = allPts.reduce((s, p) => s + p[1], 0) / allPts.length;

      if (kakaoMap) {
        kakaoMap.setCenter(new kakao.maps.LatLng(avgLat, avgLng));
        kakaoMap.setLevel(7);
      } else {
        kakaoMap = new kakao.maps.Map(container, {
          center: new kakao.maps.LatLng(avgLat, avgLng), level: 7
        });
      }

      // 기존 폴리라인 제거
      routePolylines.forEach(group => group.forEach(p => p.setMap(null)));
      routePolylines = [];
      routeOnStates = [];

      // 사용자 유형별 노선 그리기
      config.forEach((r, i) => {
        const polys = [];
        const mainPoly = new kakao.maps.Polyline({
          path: r.path.map(([lat, lng]) => new kakao.maps.LatLng(lat, lng)),
          strokeWeight: r.weight, strokeColor: r.color, strokeOpacity: 0.92, strokeStyle: r.style
        });
        mainPoly.setMap(kakaoMap);
        polys.push(mainPoly);
        if (r.busPath) {
          const busPoly = new kakao.maps.Polyline({
            path: r.busPath.map(([lat, lng]) => new kakao.maps.LatLng(lat, lng)),
            strokeWeight: 6, strokeColor: '#f97316', strokeOpacity: 0.88, strokeStyle: 'shortdot'
          });
          busPoly.setMap(kakaoMap);
          polys.push(busPoly);
        }
        routePolylines.push(polys);
        routeOnStates.push(true);
      });

      // ── 장소 핀 ──
      placeOverlays.forEach(o => o.setMap(null));
      placeOverlays = [];
      PLACES.forEach((p, idx) => {
        const coord = PLACE_COORDS[idx];
        const overlay = new kakao.maps.CustomOverlay({
          position: new kakao.maps.LatLng(coord.lat, coord.lng),
          content: makePinHTML(idx),
          xAnchor: 0.5, yAnchor: 1.0, zIndex: 3
        });
        overlay.setMap(kakaoMap);
        placeOverlays.push(overlay);
      });

      // ── 거점 핀 ──
      waypointOverlays.forEach(o => o.setMap(null));
      waypointOverlays = [];
      WAYPOINT_COORDS.forEach(wp => {
        const overlay = new kakao.maps.CustomOverlay({
          position: new kakao.maps.LatLng(wp.lat, wp.lng),
          content: makeWaypointHTML(wp),
          xAnchor: 0.5, yAnchor: 1.0, zIndex: 5
        });
        overlay.setMap(kakaoMap);
        waypointOverlays.push(overlay);
      });

      kakao.maps.event.addListener(kakaoMap, 'click', () => {
        closeWpPopup();
        if (sheetOpen) closeSheet();
      });

      renderRoutePanel();
      buildMapInfoPanel();

      // 설문에서 선택한 pref 기반으로 초기 카테고리 칩 적용
      applyInitialFilter();

      // 콘텍스트 푸시 비활성화 (현재 노출 안 함)
      // setTimeout(showPush, 4000);

      closeSheet();
      mapReady = true;
    }

    function setCat(el) {
      activeFilter = el.dataset.filter || 'all';
      document.querySelectorAll('.cat-chip').forEach(b => {
        b.classList.remove('on');
        b.setAttribute('aria-pressed', 'false');
      });
      el.classList.add('on');
      el.setAttribute('aria-pressed', 'true');

      if (currentPlace > -1 && !matchesActiveFilter(PLACES[currentPlace])) closeSheet();

      // 카카오 오버레이 표시/숨김
      placeOverlays.forEach((overlay, idx) => {
        overlay.setMap(matchesActiveFilter(PLACES[idx]) ? kakaoMap : null);
      });

      // 카드 표시/숨김
      document.querySelectorAll('.place-card[id^="c-"]').forEach(card => {
        const idx = Number(card.id.slice(2));
        card.hidden = !matchesActiveFilter(PLACES[idx]);
      });
    }

    // ── 핀 클릭 → 지도 이동 ──
    function selectPlace(idx) {
      if (!matchesActiveFilter(PLACES[idx])) return;
      const p = PLACES[idx];
      const coord = PLACE_COORDS[idx];
      currentPlace = idx;

      // 지도 해당 위치로 이동
      if (kakaoMap) kakaoMap.panTo(new kakao.maps.LatLng(coord.lat, coord.lng));

      // 핀 강조
      document.querySelectorAll('.disc-bubble').forEach(b => b.classList.remove('selected'));
      const db = document.getElementById('db-' + idx);
      if (db) db.classList.add('selected');

      document.querySelectorAll('.place-card').forEach(c => c.classList.remove('sel'));
      const card = document.getElementById('c-' + idx);
      if (card) card.classList.add('sel');

      // 미니시트 채우기
      document.getElementById('ms-name').textContent = p.name;
      document.getElementById('ms-cat').textContent  = p.cat;

      const r = p.rating || 0;
      const fullStars = Math.floor(r);
      const halfStar  = r - fullStars >= 0.5 ? 1 : 0;
      const emptyStars = 5 - fullStars - halfStar;
      document.getElementById('ms-rating-stars').textContent =
        '★'.repeat(fullStars) + (halfStar ? '☆' : '') + '☆'.repeat(emptyStars);
      document.getElementById('ms-rating-val').textContent = r.toFixed(1);
      document.getElementById('ms-rcount').textContent = `(${(p.reviewCount||0).toLocaleString()})`;

      const openEl = document.getElementById('ms-open');
      openEl.textContent = p.status || '영업중';
      openEl.className   = 'ms-open-tag ' + (p.statusTone || 'open');
      document.getElementById('ms-hours').textContent = p.hours;

      document.getElementById('ms-transit-hdr').textContent = (p.transitBase || '거점') + ' 기준';
      document.getElementById('ms-walk').textContent = p.walk;
      document.getElementById('ms-bike').textContent = p.bike;
      document.getElementById('ms-bus').textContent  = p.bus;

      openSheet();
    }

    function openSheet() {
      const sheet = document.getElementById('mini-sheet');
      sheet.classList.add('open');
      sheetOpen = true;
      requestAnimationFrame(() => {
        const h = sheet.offsetHeight;
        setMapControlsBottom(h + MAP_CONTROL_BOTTOM);
      });
    }
    function closeSheet() {
      const sheet = document.getElementById('mini-sheet');
      sheet.classList.remove('open');
      sheetOpen = false;
      currentPlace = -1;
      setMapControlsBottom(MAP_CONTROL_BOTTOM);
      document.querySelectorAll('.disc-bubble').forEach(b => b.classList.remove('selected'));
      document.querySelectorAll('.place-card').forEach(c => c.classList.remove('sel'));
    }
